229 template <typename T, typename std::enable_if<std::is_integral<T>::value, std::nullptr_t>::type = nullptr>
230 bool readRandom(T& value, const T lower, const T upper)
231 {
232 static_assert(sizeof(T) <= 32, "readRandom only supports types up to 32 bytes");
233 value = lower;
234 if (upper <= lower) {
235 M5_LIB_LOGE("lower must be less than upper");
236 return false;
237 }
238
239 using U = typename std::make_unsigned<T>::type;
240 const U range = static_cast<U>(static_cast<U>(upper) - static_cast<U>(lower));
241 const U limit = std::numeric_limits<U>::max() - (std::numeric_limits<U>::max() % range);
242 uint8_t rv[32]{};
243 uint_fast8_t offset{};
244 while (true) {
245 if (!readRandomArray(rv)) {
246 return false;
247 }
248 offset = 0;
249 while (offset + sizeof(U) <= 32) {
250 U raw{};
251 memcpy(&raw, rv + offset, sizeof(U));
252 offset += sizeof(U);
253
254 if (raw >= limit) { // Rejection sampling
255 continue;
256 }
257 value = static_cast<T>(lower + (raw % range));
258 return true;
259 }
260 }
261 return false;
262 }
272 template <typename T, typename std::enable_if<std::is_floating_point<T>::value, std::nullptr_t>::type = nullptr>
273 bool readRandom(T& value, const T lower, const T upper)
274 {
275 value = std::numeric_limits<T>::quiet_NaN();
276 if (upper <= lower) {
277 M5_LIB_LOGE("lower must be less than upper");
278 return false;
279 }
280 uint8_t rv[32]{};
281 if (!readRandomArray(rv)) {
282 return false;
283 }
284 uint32_t raw{};
285 memcpy(&raw, rv, sizeof(uint32_t)); // use first 4 bytes
286
287 // convert to [0.0, 1.0)
288 using real_t = typename std::common_type<T, double>::type;
289 const real_t norm = real_t(1) / (static_cast<real_t>(std::numeric_limits<uint32_t>::max()) + real_t(1));
290 const real_t r = static_cast<real_t>(raw) * norm;
291 // Interpolate without (upper - lower), which overflows for a range wider than max (e.g. [lowest, max))
292 value = static_cast<T>(static_cast<real_t>(lower) * (real_t(1) - r) + static_cast<real_t>(upper) * r);
293 if (value >= upper) { // Rounding may reach upper
294 value = std::nextafter(upper, lower);
295 }
296 return std::isfinite(value);
297 }
