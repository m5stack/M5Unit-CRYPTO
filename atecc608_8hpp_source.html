205inline uint16_t offset_to_param2_for_config(const uint8_t offset)
206{
207 const uint8_t block = (offset >> 5) & 0x03; // 0〜3
208 const uint8_t index = ((offset & 31) >> 2) & 0x07; // 0〜7
209 return (block << 3) | index;
210}
212inline uint16_t slot_block_to_param2(const uint8_t slot, const uint16_t offset)
213{
214 const uint8_t block = offset >> 5; // 0〜12
215 const uint8_t word_offset = (offset & 31) >> 2; // 0〜7
216 return (slot << 3) | (block << 8) | word_offset;
217}

344 DateTime get_issue_date() const
345 {
346 DateTime dt{};
347 uint8_t b64 = _data[64];
348 uint8_t b65 = _data[65];
349 uint8_t b66 = _data[66];
350 uint8_t b71 = _data[71];
351 uint8_t fmt_ver = format_version();
352
353 if (fmt_ver == 1 || fmt_ver == 2) {
354 dt.tm_year = ((((b71 & 0xC0) >> 1) | ((b64 >> 3) & 0x1F)) + 100);
355 } else {
356 dt.tm_year = ((b64 >> 3) + 100);
357 }
358 dt.tm_mon = (((b64 & 0x07) << 1) | ((b65 & 0x80) >> 7)) - 1;
359 dt.tm_mday = (b65 & 0x7C) >> 2;
360 dt.tm_hour = ((b65 & 0x03) << 3) | ((b66 & 0xE0) >> 5);
361 dt.tm_min = 0;
362 dt.tm_sec = 0;
363 return dt;
364 }
365
366 DateTime get_expire_date() const
367 {
368 DateTime dt = issue_date();
369 uint8_t b66 = _data[66];
370 uint8_t b71 = _data[71];
371 uint8_t fmt_ver = format_version();
372 uint8_t expire_years{};
373
374 if (fmt_ver == 1 || fmt_ver == 2) {
375 expire_years = (b66 & 0x1F) | ((b71 & 0x30) << 1);
376 } else {
377 expire_years = b66 & 0x1F;
378 }
379
380 if (expire_years != 0) {
381 dt.tm_year += expire_years;
382 } else {
383 // indefinite
384 dt.tm_year = 9999 - 1900;
385 dt.tm_mon = 11;
386 dt.tm_mday = 31;
387 dt.tm_hour = 23;
388 dt.tm_min = 59;
389 dt.tm_sec = 59;
390 }
391 return dt;
392 }
